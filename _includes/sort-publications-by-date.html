{% comment %} Merges post-mortem pages and published articles, then orders by epoch (newest first). Requires include.postmortems and include.articles arrays. Sets publication_sorted_urls (array of document url strings, newest first). {% endcomment %} {% assign _pool = include.postmortems | concat: include.articles %} {% assign _unsorted = _pool %} {% assign _sorted_urls = "" %} {% for _pick in (1.._pool.size) %} {% assign _best_url = "" %} {% assign _best_epoch = 0 %} {% for _item in _unsorted %} {% assign _ep = _item.date | date: '%s' | plus: 0 %} {% if _ep > _best_epoch %} {% assign _best_epoch = _ep %} {% assign _best_url = _item.url %} {% endif %} {% endfor %} {% if _sorted_urls == "" %} {% assign _sorted_urls = _best_url %} {% else %} {% assign _sorted_urls = _sorted_urls | append: "|" | append: _best_url %} {% endif %} {% assign _unsorted = _unsorted | where_exp: "_item", "_item.url != _best_url" %} {% endfor %} {% assign publication_sorted_urls = _sorted_urls | split: "|" %}
